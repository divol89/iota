4//! Historical view queries: exact id+version lookups without consistency
5//! filtering. Combines all objects from `checkpointed_objects` with all
6//! past versions from `objects_backward_history`.
7
8use crate::{
9    backward_view::{
10        BACKWARD_HISTORY_WATERMARK_ENTITY, CHECKPOINTED_COLUMNS, HISTORY_COLUMNS, HistoricalFilter,
11        NOT_YET_CREATED, NativeObjectStatus, merge_and_deduplicate,
12    },
13    filter, query,
14    raw_query::RawQuery,
15    types::{
16        cursor::Page,
17        object::{Cursor, StoredBackwardObject},
18    },
19};
20
21/// Builds a historical view query for a `HistoricalFilter`. Internally
22/// branches on whether type/owner are also constrained: keys-only filters
23/// include the `objects_version` source so real tombstone versions are
24/// reachable, while filters with type/owner skip it (tombstones can't
25/// match).
26pub(crate) fn query(
27    checkpoint_viewed_at: u64,
28    page: &Page<Cursor>,
29    filter: &HistoricalFilter,
30) -> RawQuery {
31    let filter_fn = |q| filter.apply(q);
32
33    let mut sources = vec![
34        checkpointed_objects(page, &filter_fn),
35        historical_objects(page, &filter_fn),
36    ];
37    if !filter.has_type_or_owner() {
38        sources.push(tombstones_from_objects_version(
39            checkpoint_viewed_at,
40            page,
41            &filter_fn,
42        ));
43    }
44    merge_and_deduplicate(sources)
45}
46
47/// Returns all objects from `checkpointed_objects` (including tombstones)
48/// that satisfy the provided filter.
49fn checkpointed_objects(
50    page: &Page<Cursor>,
51    filter_fn: &impl Fn(RawQuery) -> RawQuery,
52) -> RawQuery {
53    let checkpointed_filtered = filter_fn(query!(format!(
54        "SELECT {CHECKPOINTED_COLUMNS} FROM checkpointed_objects"
55    )));
56    let source = query!(
57        "SELECT candidates.* FROM ({}) candidates",
58        checkpointed_filtered
59    );
60    page.apply::<StoredBackwardObject>(source)
61}
62
63/// Returns all objects from `objects_backward_history` that satisfy the
64/// provided filter, excluding `NOT_YET_CREATED` entries.
65fn historical_objects(page: &Page<Cursor>, filter_fn: &impl Fn(RawQuery) -> RawQuery) -> RawQuery {
66    let history_filtered = filter_fn(query!(format!(
67        "SELECT {HISTORY_COLUMNS} FROM objects_backward_history"
68    )));
69    let history_window = filter!(
70        history_filtered,
71        format!("object_status != {NOT_YET_CREATED}")
72    );
73    let source = query!("SELECT candidates.* FROM ({}) candidates", history_window);
74    page.apply::<StoredBackwardObject>(source)
75}

77/// Returns synthetic `WrappedOrDeleted` tombstone rows from `objects_version`
78/// for versions that exist there but are NOT present in `checkpointed_objects`
79/// or `objects_backward_history`. This allows `objectKeys` lookups to find
80/// objects by their real tombstone version.
81///
82/// Why surviving rows are necessarily `WrappedOrDeleted`: every version in
83/// `objects_version` falls into exactly one of three cases.
84///   1. Currently-latest state of the object (active row or tombstone) — lives
85///      in `checkpointed_objects` and is filtered out by the first `NOT
86///      EXISTS`.
87///   2. Prior active state superseded by a later transaction — recorded in
88///      `objects_backward_history` at its actual `object_version` with status
89///      `ACTIVE`, and filtered out by the second `NOT EXISTS`.
90///   3. Prior wrap/delete tombstone that was later overwritten in
91///      `checkpointed_objects` (typically by an unwrap). The wrap state is
92///      recorded in `objects_backward_history`, but at `lamport - 1` of the
93///      unwrapping transaction — which generally differs from the real
94///      tombstone version. The real version in `objects_version` therefore
95///      survives both `NOT EXISTS` filters. By construction this is the only
96///      surviving case, hence the static `WrappedOrDeleted` tag.
97///
98/// Only used for keys-only queries where the filter contains only
99/// `(object_id, object_version)` pairs — the `NOT EXISTS` subqueries hit
100/// primary keys so the cost is proportional to the number of requested keys.
101///
102/// Filters out versions outside the `[backward-history watermark,
103/// checkpoint_viewed_at]` window to avoid false tombstones from pruned ranges
104/// and concurrent in-flight batch writes.
105fn tombstones_from_objects_version(
106    checkpoint_viewed_at: u64,
107    page: &Page<Cursor>,
108    filter_fn: &impl Fn(RawQuery) -> RawQuery,
109) -> RawQuery {
110    let wrapped_or_deleted = NativeObjectStatus::WrappedOrDeleted as i16;
111    let checkpoint_viewed_at = checkpoint_viewed_at as i64;
112
113    // Apply the keys filter directly to the bare table select so its disjunctive
114    // `(object_id = ... AND object_version = ...)` clauses become the leading
115    // WHERE — this lets the planner use the `(object_id, object_version)`
116    // primary key for the lookup instead of scanning the whole cp window. The
117    // cp bounds and the two `NOT EXISTS` checks are then AND'd on top.
118    let base = query!(format!(
119        "SELECT object_id, object_version, \
120         {wrapped_or_deleted}::smallint AS object_status, \
121         NULL::bytea AS object_digest, \
122         NULL::smallint AS owner_type, \
123         NULL::bytea AS owner_id, \
124         NULL::text AS object_type, \
125         NULL::bytea AS object_type_package, \
126         NULL::text AS object_type_module, \
127         NULL::text AS object_type_name, \
128         NULL::bytea AS serialized_object, \
129         NULL::text AS coin_type, \
130         NULL::bigint AS coin_balance, \
131         NULL::smallint AS df_kind, \
132         FALSE AS from_backward_history \
133         FROM objects_version ov"
134    ));
135
136    let with_keys = filter_fn(base);
137
138    let with_bounds = filter!(
139        with_keys,
140        format!(
141            "cp_sequence_number >= COALESCE(\
142                 (SELECT min_available_cp FROM watermarks \
143                  WHERE entity = '{BACKWARD_HISTORY_WATERMARK_ENTITY}'), 0) \
144             AND cp_sequence_number <= {checkpoint_viewed_at}"
145        )
146    );
147
148    let inner = filter!(
149        with_bounds,
150        "NOT EXISTS (\
151             SELECT 1 FROM checkpointed_objects co \
152             WHERE co.object_id = ov.object_id \
153               AND co.object_version = ov.object_version) \
154         AND NOT EXISTS (\
155             SELECT 1 FROM objects_backward_history bh \
156             WHERE bh.object_id = ov.object_id \
157               AND bh.object_version = ov.object_version)"
158    );
159
160    let source = query!("SELECT candidates.* FROM ({}) candidates", inner);
161    page.apply::<StoredBackwardObject>(source)
162}
