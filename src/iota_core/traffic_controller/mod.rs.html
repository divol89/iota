47type Blocklist = Arc<DashMap<IpAddr, SystemTime>>;

49#[derive(Clone)]
50pub struct Blocklists {

52    proxied_clients: Blocklist,
53}

55#[derive(Clone)]
56pub enum Acl {
57    Blocklists(Blocklists),
58    /// If this variant is set, then we do no tallying or running
59    /// of background tasks, and instead simply block all IPs not
60    /// in the allowlist on calls to `check`. The allowlist should
61    /// only be populated once at initialization.
62    Allowlist(Vec<IpAddr>),
63}

74}

76impl Debug for TrafficController {
77    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
78        // NOTE: we do not want to print the contents of the blocklists to logs
79        // given that (1) it contains all requests IPs, and (2) it could be quite
80        // large. Instead, we print lengths of the blocklists. Further, we prefer
81        // to get length from the metrics rather than from the blocklists themselves
82        // to avoid unnecessarily acquiring the read lock.
83        f.debug_struct("TrafficController")
84            .field(
85                "connection_ip_blocklist_len",
86                &self.metrics.connection_ip_blocklist_len.get(),
87            )
88            .field(
89                "proxy_ip_blocklist_len",
90                &self.metrics.proxy_ip_blocklist_len.get(),
91            )
92            .finish()
93    }
94}

97    pub async fn init(
98        policy_config: PolicyConfig,
99        metrics: Arc<TrafficControllerMetrics>,
100        fw_config: Option<RemoteFirewallConfig>,
101    ) -> Self {
102        metrics.dry_run_enabled.set(policy_config.dry_run as i64);
103        match policy_config.allow_list.clone() {
104            Some(allow_list) => {
105                let allowlist = allow_list
106                    .into_iter()
107                    .map(|ip_str| {
108                        parse_ip(&ip_str).unwrap_or_else(|| {
109                            fatal!("Failed to parse allowlist IP address: {ip_str:?}")
110                        })
111                    })
112                    .collect();
113                Self {
114                    tally_channel: Arc::new(ParkingLotMutex::new(None)),
115                    acl: Acl::Allowlist(allowlist),
116                    metrics,
117                    policy_config: Arc::new(RwLock::new(policy_config)),
118                    fw_config,
119                    spam_policy: None,
120                    error_policy: None,
121                }
122            }
123            None => {
124                let spam_policy = Arc::new(Mutex::new(
125                    TrafficControlPolicy::from_spam_config(policy_config.clone()).await,
126                ));
127                let error_policy = Arc::new(Mutex::new(
128                    TrafficControlPolicy::from_error_config(policy_config.clone()).await,
129                ));
130                let this = Self {
131                    tally_channel: Arc::new(ParkingLotMutex::new(None)),
132                    acl: Acl::Blocklists(Blocklists {
133                        clients: Arc::new(DashMap::new()),
134                        proxied_clients: Arc::new(DashMap::new()),
135                    }),
136                    metrics,
137                    policy_config: Arc::new(RwLock::new(policy_config)),
138                    fw_config,
139                    spam_policy: Some(spam_policy),
140                    error_policy: Some(error_policy),
141                };
142                this.spawn().await;
143                this
144            }
145        }
146    }

156    async fn spawn(&self) {
157        let policy_config = { self.policy_config.read().await.clone() };
158        Self::set_policy_config_metrics(&policy_config, self.metrics.clone());
159        let (tx, rx) = mpsc::channel(policy_config.channel_capacity);
160        // Memoized drainfile existence state. This is passed into delegation
161        // functions to prevent them from continuing to populate blocklists
162        // if drain is set, as otherwise it will grow without bounds
163        // without the firewall running to periodically clear it.
164        let mem_drainfile_present = self
165            .fw_config
166            .as_ref()
167            .map(|config| config.drain_path.exists())
168            .unwrap_or(false);
169        self.metrics
170            .deadmans_switch_enabled
171            .set(mem_drainfile_present as i64);
172        let blocklists = match self.acl.clone() {
173            Acl::Blocklists(blocklists) => blocklists,
174            Acl::Allowlist(_) => fatal!("Allowlist ACL should not exist on spawn"),
175        };
176        let tally_loop_blocklists = blocklists.clone();
177        let tally_loop_metrics = self.metrics.clone();
178        let clear_loop_metrics = self.metrics.clone();
179        let tally_loop_fw_config = self.fw_config.clone();
180
181        let spam_policy = self
182            .spam_policy
183            .clone()
184            .expect("spam policy should exist on spawn");
185        let error_policy = self
186            .error_policy
187            .clone()
188            .expect("error policy should exist on spawn");
189
190        spawn_monitored_task!(run_tally_loop(
191            rx,
192            policy_config,
193            spam_policy,
194            error_policy,
195            tally_loop_fw_config,
196            tally_loop_blocklists,
197            tally_loop_metrics,
198            mem_drainfile_present,
199        ));
200        spawn_monitored_task!(run_clear_blocklists_loop(blocklists, clear_loop_metrics));
201        self.open_tally_channel(tx);
202    }

204    pub async fn get_current_state(&self) -> TrafficControlReconfigParams {
205        let mut result = TrafficControlReconfigParams {
206            error_threshold: None,
207            spam_threshold: None,
208            dry_run: None,
209        };
210
211        if let Some(error_policy) = self.error_policy.as_ref() {
212            if let TrafficControlPolicy::FreqThreshold(ref policy) = *error_policy.lock().await {
213                result.error_threshold = Some(policy.client_threshold);
214            }
215        }
216
217        if let Some(spam_policy) = self.spam_policy.as_ref() {
218            if let TrafficControlPolicy::FreqThreshold(ref policy) = *spam_policy.lock().await {
219                result.spam_threshold = Some(policy.client_threshold);
220            }
221        }
222
223        result.dry_run = Some(self.policy_config.read().await.dry_run);
224        result
225    }

227    pub async fn admin_reconfigure(
228        &self,
229        params: TrafficControlReconfigParams,
230    ) -> Result<TrafficControlReconfigParams, IotaError> {
231        let TrafficControlReconfigParams {
232            error_threshold,
233            spam_threshold,
234            dry_run,
235        } = params;
236        if let Some(error_threshold) = error_threshold {
237            self.metrics
238                .error_client_threshold
239                .set(error_threshold as i64);
240            Self::update_policy_threshold(
241                self.error_policy.as_ref().unwrap(),
242                error_threshold,
243                dry_run,
244            )
245            .await?;
246        }
247        if let Some(spam_threshold) = spam_threshold {
248            self.metrics
249                .spam_client_threshold
250                .set(spam_threshold as i64);
251            Self::update_policy_threshold(
252                self.spam_policy.as_ref().unwrap(),
253                spam_threshold,
254                dry_run,
255            )
256            .await?;
257        }
258        if let Some(dry_run) = dry_run {
259            self.metrics.dry_run_enabled.set(dry_run as i64);
260            self.policy_config.write().await.dry_run = dry_run;
261        }
262
263        Ok(self.get_current_state().await)
264    }

266    async fn update_policy_threshold(
267        policy: &Arc<Mutex<TrafficControlPolicy>>,
268        threshold: u64,
269        dry_run: Option<bool>,
270    ) -> Result<(), IotaError> {
271        match *policy.lock().await {
272            TrafficControlPolicy::FreqThreshold(ref mut policy) => {
273                policy.client_threshold = threshold;
274                if let Some(dry_run) = dry_run {
275                    policy.config.dry_run = dry_run;
276                }
277                Ok(())
278            }
279            TrafficControlPolicy::TestNConnIP(ref mut policy) => {
280                policy.threshold = threshold;
281                if let Some(dry_run) = dry_run {
282                    policy.config.dry_run = dry_run;
283                }
284                Ok(())
285            }
286            _ => Err(IotaError::InvalidAdminRequest(
287                "Unsupported prior policy type during traffic control reconfiguration".to_string(),
288            )),
289        }
290    }

318    pub fn tally(&self, tally: TrafficTally) {
319        if let Some(channel) = self.tally_channel.lock().as_ref() {
320            // Use try_send rather than send mainly to avoid creating backpressure
321            // on the caller if the channel is full, which may slow down the critical
322            // path. Dropping the tally on the floor should be ok, as in this case
323            // we are effectively sampling traffic, which we would need to do anyway
324            // if we are overloaded
325            match channel.try_send(tally) {
326                Err(TrySendError::Full(_)) => {
327                    warn!("TrafficController tally channel full, dropping tally");
328                    self.metrics.tally_channel_overflow.inc();
329                    // TODO: once we've verified this doesn't happen under
330                    // normal conditions, we can consider
331                    // dropping the request itself given
332                    // that clearly the system is overloaded
333                }
334                Err(TrySendError::Closed(_)) => {
335                    warn!("TrafficController tally channel closed unexpectedly");
336                }
337                Ok(_) => {}
338            }
339        } else {
340            warn!("TrafficController not yet accepting tally requests.");
341        }
342    }

344    /// Handle check with dry-run mode considered
345    pub async fn check(&self, client: &Option<IpAddr>, proxied_client: &Option<IpAddr>) -> bool {
346        let policy_config = { self.policy_config.read().await.clone() };
347        let check_with_dry_run_maybe = |allowed| -> bool {
348            match (allowed, policy_config.dry_run) {
349                // request allowed
350                (true, _) => true,
351                // request blocked while in dry-run mode
352                (false, true) => {
353                    debug!("Dry run mode: Blocked request from client {:?}", client);
354                    self.metrics.num_dry_run_blocked_requests.inc();
355                    true
356                }
357                // request blocked
358                (false, false) => {
359                    debug!("Blocked request from client {:?}", client);
360                    self.metrics.requests_blocked_at_protocol.inc();
361                    false
362                }
363            }
364        };
365
366        match &self.acl {
367            Acl::Allowlist(allowlist) => {
368                let allowed = client.is_none() || allowlist.contains(&client.unwrap());
369                check_with_dry_run_maybe(allowed)
370            }
371            Acl::Blocklists(blocklists) => {
372                let allowed = self
373                    .check_blocklists(blocklists, client, proxied_client)
374                    .await;
375                check_with_dry_run_maybe(allowed)
376            }
377        }
378    }

380    /// Returns true if the connection is in blocklist, false otherwise
381    async fn check_blocklists(
382        &self,
383        blocklists: &Blocklists,
384        client: &Option<IpAddr>,
385        proxied_client: &Option<IpAddr>,
386    ) -> bool {
387        let client_check = self.check_and_clear_blocklist(
388            client,
389            blocklists.clients.clone(),
390            &self.metrics.connection_ip_blocklist_len,
391        );
392        let proxied_client_check = self.check_and_clear_blocklist(
393            proxied_client,
394            blocklists.proxied_clients.clone(),
395            &self.metrics.proxy_ip_blocklist_len,
396        );
397        let (client_check, proxied_client_check) =
398            futures::future::join(client_check, proxied_client_check).await;
399        client_check && proxied_client_check
400    }
401
402    async fn check_and_clear_blocklist(
403        &self,
404        client: &Option<IpAddr>,
405        blocklist: Blocklist,
406        blocklist_len_gauge: &IntGauge,
407    ) -> bool {
408        let client = match client {
409            Some(client) => client,
410            None => return true,
411        };
412        let now = SystemTime::now();
413        // the below two blocks cannot be nested, otherwise we will deadlock
414        // due to acquiring the lock on get, then holding across the remove
415        let (should_block, should_remove) = {
416            match blocklist.get(client) {
417                Some(expiration) if now >= *expiration => (false, true),
418                None => (false, false),
419                _ => (true, false),
420            }
421        };
422        if should_remove && blocklist.remove(client).is_some() {
423            blocklist_len_gauge.dec();
424        }
425        !should_block
426    }
427}

429/// Although we clear IPs from the blocklist lazily when they are checked,
430/// it's possible that over time we may accumulate a large number of stale
431/// IPs in the blocklist for clients that are added, then once blocked,
432/// never checked again. This function runs periodically to clear out any
433/// such stale IPs. This also ensures that the blocklist length metric
434/// accurately reflects TTL.
435async fn run_clear_blocklists_loop(blocklists: Blocklists, metrics: Arc<TrafficControllerMetrics>) {
436    loop {
437        tokio::time::sleep(Duration::from_secs(3)).await;
438        let now = SystemTime::now();
439        blocklists.clients.retain(|_, expiration| now < *expiration);
440        blocklists
441            .proxied_clients
442            .retain(|_, expiration| now < *expiration);
443        metrics
444            .connection_ip_blocklist_len
445            .set(blocklists.clients.len() as i64);
446        metrics
447            .proxy_ip_blocklist_len
448            .set(blocklists.proxied_clients.len() as i64);
449    }
450}

452async fn run_tally_loop(
453    mut receiver: mpsc::Receiver<TrafficTally>,
454    policy_config: PolicyConfig,
455    spam_policy: Arc<Mutex<TrafficControlPolicy>>,
456    error_policy: Arc<Mutex<TrafficControlPolicy>>,
457    fw_config: Option<RemoteFirewallConfig>,
458    blocklists: Blocklists,
459    metrics: Arc<TrafficControllerMetrics>,
460    mut mem_drainfile_present: bool,
461) {
462    let spam_blocklists = Arc::new(blocklists.clone());
463    let error_blocklists = Arc::new(blocklists);
464    let node_fw_client = fw_config
465        .as_ref()
466        .map(|fw_config| NodeFWClient::new(fw_config.remote_fw_url.clone()));
467
468    let timeout = fw_config
469        .as_ref()
470        .map(|fw_config| fw_config.drain_timeout_secs)
471        .unwrap_or(DEFAULT_DRAIN_TIMEOUT_SECS);
472    let mut metric_timer = Instant::now();
473
474    loop {
475        tokio::select! {
476            received = receiver.recv() => {
477                metrics.tallies.inc();
478                match received {
479                    Some(tally) => {
480                        // TODO: spawn a task to handle tallying concurrently
481                        if let Err(err) = handle_spam_tally(
482                            spam_policy.clone(),
483                            &policy_config,
484                            &node_fw_client,
485                            &fw_config,
486                            tally.clone(),
487                            spam_blocklists.clone(),
488                            metrics.clone(),
489                            mem_drainfile_present,
490                        )
491                        .await {
492                            warn!("Error handling spam tally: {}", err);
493                        }
494                        if let Err(err) = handle_error_tally(
495                            error_policy.clone(),
496                            &policy_config,
497                            &node_fw_client,
498                            &fw_config,
499                            tally,
500                            error_blocklists.clone(),
501                            metrics.clone(),
502                            mem_drainfile_present,
503                        )
504                        .await {
505                            warn!("Error handling error tally: {}", err);
506                        }
507                    }
508                    None => {
509                        info!("TrafficController tally channel closed by all senders");
510                        return;
511                    }
512                }
513            }
514            // Dead man's switch - if we suspect something is sinking all traffic to node, disable nodefw
515            _ = tokio::time::sleep(tokio::time::Duration::from_secs(timeout)) => {
516                if let Some(fw_config) = &fw_config {
517                    error!("No traffic tallies received in {} seconds.", timeout);
518                    if mem_drainfile_present {
519                        continue;
520                    }
521                    if !fw_config.drain_path.exists() {
522                        mem_drainfile_present = true;
523                        warn!("Draining Node firewall.");
524                        File::create(&fw_config.drain_path)
525                            .expect("Failed to touch nodefw drain file");
526                        metrics.deadmans_switch_enabled.set(1);
527                    }
528                }
529            }
530        }
531
532        // every N seconds, we update metrics and logging that would be too
533        // spammy to be handled while processing each tally
534        if metric_timer.elapsed() > Duration::from_secs(METRICS_INTERVAL_SECS) {
535            if let TrafficControlPolicy::FreqThreshold(ref spam_policy) = *spam_policy.lock().await
536            {
537                if let Some(highest_direct_rate) = spam_policy.highest_direct_rate() {
538                    metrics
539                        .highest_direct_spam_rate
540                        .set(highest_direct_rate.0 as i64);
541                    debug!("Recent highest direct spam rate: {:?}", highest_direct_rate);
542                }
543                if let Some(highest_proxied_rate) = spam_policy.highest_proxied_rate() {
544                    metrics
545                        .highest_proxied_spam_rate
546                        .set(highest_proxied_rate.0 as i64);
547                    debug!(
548                        "Recent highest proxied spam rate: {:?}",
549                        highest_proxied_rate
550                    );
551                }
552            }
553            if let TrafficControlPolicy::FreqThreshold(ref error_policy) =
554                *error_policy.lock().await
555            {
556                if let Some(highest_direct_rate) = error_policy.highest_direct_rate() {
557                    metrics
558                        .highest_direct_error_rate
559                        .set(highest_direct_rate.0 as i64);
560                    debug!(
561                        "Recent highest direct error rate: {:?}",
562                        highest_direct_rate
563                    );
564                }
565                if let Some(highest_proxied_rate) = error_policy.highest_proxied_rate() {
566                    metrics
567                        .highest_proxied_error_rate
568                        .set(highest_proxied_rate.0 as i64);
569                    debug!(
570                        "Recent highest proxied error rate: {:?}",
571                        highest_proxied_rate
572                    );
573                }
574            }
575            metric_timer = Instant::now();
576        }
577    }
578}

580async fn handle_error_tally(
581    policy: Arc<Mutex<TrafficControlPolicy>>,
582    policy_config: &PolicyConfig,
583    nodefw_client: &Option<NodeFWClient>,
584    fw_config: &Option<RemoteFirewallConfig>,
585    tally: TrafficTally,
586    blocklists: Arc<Blocklists>,
587    metrics: Arc<TrafficControllerMetrics>,
588    mem_drainfile_present: bool,
589) -> Result<(), reqwest::Error> {
590    let Some((error_weight, error_type)) = tally.clone().error_info else {
591        return Ok(());
592    };
593    if !error_weight.is_sampled() {
594        return Ok(());
595    }
596    trace!(
597        "Handling error_type {:?} from client {:?}",
598        error_type, tally.direct,
599    );
600    metrics
601        .tally_error_types
602        .with_label_values(&[error_type.as_str()])
603        .inc();
604    let resp = policy.lock().await.handle_tally(tally);
605    metrics.error_tally_handled.inc();
606    if let Some(fw_config) = fw_config {
607        if fw_config.delegate_error_blocking && !mem_drainfile_present {
608            let client = nodefw_client
609                .as_ref()
610                .expect("Expected NodeFWClient for blocklist delegation");
611            return delegate_policy_response(
612                resp,
613                policy_config,
614                client,
615                fw_config.destination_port,
616                metrics.clone(),
617            )
618            .await;
619        }
620    }
621    handle_policy_response(resp, policy_config, blocklists, metrics).await;
622    Ok(())
623}

625async fn handle_spam_tally(
626    policy: Arc<Mutex<TrafficControlPolicy>>,
627    policy_config: &PolicyConfig,
628    nodefw_client: &Option<NodeFWClient>,
629    fw_config: &Option<RemoteFirewallConfig>,
630    tally: TrafficTally,
631    blocklists: Arc<Blocklists>,
632    metrics: Arc<TrafficControllerMetrics>,
633    mem_drainfile_present: bool,
634) -> Result<(), reqwest::Error> {
635    if !(tally.spam_weight.is_sampled() && policy_config.spam_sample_rate.is_sampled()) {
636        return Ok(());
637    }
638    let resp = policy.lock().await.handle_tally(tally.clone());
639    metrics.tally_handled.inc();
640    if let Some(fw_config) = fw_config {
641        if fw_config.delegate_spam_blocking && !mem_drainfile_present {
642            let client = nodefw_client
643                .as_ref()
644                .expect("Expected NodeFWClient for blocklist delegation");
645            return delegate_policy_response(
646                resp,
647                policy_config,
648                client,
649                fw_config.destination_port,
650                metrics.clone(),
651            )
652            .await;
653        }
654    }
655    handle_policy_response(resp, policy_config, blocklists, metrics).await;
656    Ok(())
657}

659async fn handle_policy_response(
660    response: PolicyResponse,
661    policy_config: &PolicyConfig,
662    blocklists: Arc<Blocklists>,
663    metrics: Arc<TrafficControllerMetrics>,
664) {
665    let PolicyResponse {
666        block_client,
667        block_proxied_client,
668    } = response;
669    let PolicyConfig {
670        connection_blocklist_ttl_sec,
671        proxy_blocklist_ttl_sec,
672        ..
673    } = policy_config;
674    if let Some(client) = block_client {
675        if blocklists
676            .clients
677            .insert(
678                client,
679                SystemTime::now() + Duration::from_secs(*connection_blocklist_ttl_sec),
680            )
681            .is_none()
682        {
683            // Only increment the metric if the client was not already blocked
684            debug!("Adding client {:?} to blocklist", client);
685            metrics.connection_ip_blocklist_len.inc();
686        }
687    }
688    if let Some(client) = block_proxied_client {
689        if blocklists
690            .proxied_clients
691            .insert(
692                client,
693                SystemTime::now() + Duration::from_secs(*proxy_blocklist_ttl_sec),
694            )
695            .is_none()
696        {
697            // Only increment the metric if the client was not already blocked
698            debug!("Adding proxied client {:?} to blocklist", client);
699            metrics.proxy_ip_blocklist_len.inc();
700        }
701    }
702}
703
704async fn delegate_policy_response(
705    response: PolicyResponse,
706    policy_config: &PolicyConfig,
707    node_fw_client: &NodeFWClient,
708    destination_port: u16,
709    metrics: Arc<TrafficControllerMetrics>,
710) -> Result<(), reqwest::Error> {
711    let PolicyResponse {
712        block_client,
713        block_proxied_client,
714    } = response;
715    let PolicyConfig {
716        connection_blocklist_ttl_sec,
717        proxy_blocklist_ttl_sec,
718        ..
719    } = policy_config;
720    let mut addresses = vec![];
721    if let Some(client_id) = block_client {
722        debug!("Delegating client blocking to firewall");
723        addresses.push(BlockAddress {
724            source_address: client_id.to_string(),
725            destination_port,
726            ttl: *connection_blocklist_ttl_sec,
727        });
728    }
729    if let Some(ip) = block_proxied_client {
730        debug!("Delegating proxied client blocking to firewall");
731        addresses.push(BlockAddress {
732            source_address: ip.to_string(),
733            destination_port,
734            ttl: *proxy_blocklist_ttl_sec,
735        });
736    }
737    if addresses.is_empty() {
738        Ok(())
739    } else {
740        metrics
741            .blocks_delegated_to_firewall
742            .inc_by(addresses.len() as u64);
743        match node_fw_client
744            .block_addresses(BlockAddresses { addresses })
745            .await
746        {
747            Ok(()) => Ok(()),
748            Err(err) => {
749                metrics.firewall_delegation_request_fail.inc();
750                Err(err)
751            }
752        }
753    }
754}

756#[derive(Debug, Clone)]
757pub struct TrafficSimMetrics {
758    pub num_requests: u64,
759    pub num_blocked: u64,
760    pub time_to_first_block: Option<Duration>,
761    pub abs_time_to_first_block: Option<Duration>,
762    pub total_time_blocked: Duration,
763    pub num_blocklist_adds: u64,
764}
765
766impl Default for TrafficSimMetrics {
767    fn default() -> Self {
768        Self {
769            num_requests: 0,
770            num_blocked: 0,
771            time_to_first_block: None,
772            abs_time_to_first_block: None,
773            total_time_blocked: Duration::from_micros(0),
774            num_blocklist_adds: 0,
775        }
776    }
777}
778
779impl Add for TrafficSimMetrics {
780    type Output = Self;
781
782    fn add(self, other: Self) -> Self {
783        Self {
784            num_requests: self.num_requests + other.num_requests,
785            num_blocked: self.num_blocked + other.num_blocked,
786            time_to_first_block: match (self.time_to_first_block, other.time_to_first_block) {
787                (Some(a), Some(b)) => Some(a + b),
788                (Some(a), None) => Some(a),
789                (None, Some(b)) => Some(b),
790                (None, None) => None,
791            },
792            abs_time_to_first_block: match (
793                self.abs_time_to_first_block,
794                other.abs_time_to_first_block,
795            ) {
796                (Some(a), Some(b)) => Some(a.min(b)),
797                (Some(a), None) => Some(a),
798                (None, Some(b)) => Some(b),
799                (None, None) => None,
800            },
801            total_time_blocked: self.total_time_blocked + other.total_time_blocked,
802            num_blocklist_adds: self.num_blocklist_adds + other.num_blocklist_adds,
803        }
804    }
805}

811impl TrafficSim {
812    pub async fn run(
813        policy: PolicyConfig,
814        num_clients: u8,
815        per_client_tps: usize,
816        duration: Duration,
817        report: bool,
818    ) -> TrafficSimMetrics {
819        assert!(
820            per_client_tps <= 10_000,
821            "per_client_tps must be less than 10,000. For higher values, increase num_clients"
822        );
823        assert!(num_clients < 20, "num_clients must be greater than 0");
824        assert!(num_clients > 0);
825        assert!(per_client_tps > 0);
826        assert!(duration.as_secs() > 0);
827
828        let controller = TrafficController::init_for_test(policy.clone(), None).await;
829        let tasks = (0..num_clients).map(|task_num| {
830            tokio::spawn(Self::run_single_client(
831                controller.clone(),
832                duration,
833                task_num,
834                per_client_tps,
835            ))
836        });
837
838        let status_task = if report {
839            Some(tokio::spawn(async move {
840                println!(
841                    "Running naive traffic simulation for {} seconds",
842                    duration.as_secs()
843                );
844                println!("Policy: {policy:#?}");
845                println!("Num clients: {num_clients}");
846                println!("TPS per client: {per_client_tps}");
847                println!(
848                    "Target total TPS: {}",
849                    per_client_tps * num_clients as usize
850                );
851                println!("\n");
852                for _ in 0..duration.as_secs() {
853                    print!(".");
854                    tokio::time::sleep(Duration::from_secs(1)).await;
855                }
856                println!();
857            }))
858        } else {
859            None
860        };
861
862        let metrics = futures::future::join_all(tasks).await.into_iter().fold(
863            TrafficSimMetrics::default(),
864            |acc, run_client_ret| match run_client_ret {
865                Ok(metrics) => acc + metrics,
866                Err(err) => {
867                    error!("Error running traffic sim client: {:?}", err);
868                    acc
869                }
870            },
871        );
872
873        if report {
874            status_task.unwrap().await.unwrap();
875            Self::report_metrics(metrics.clone(), duration, per_client_tps, num_clients);
876        }
877        metrics
878    }
879
880    async fn run_single_client(
881        controller: TrafficController,
882        duration: Duration,
883        task_num: u8,
884        per_client_tps: usize,
885    ) -> TrafficSimMetrics {
886        // Do an initial sleep for a random amount of time to smooth
887        // out the traffic. This shouldn't be strictly necessary and
888        // we can remove if we want more determinism
889        let sleep_time = Duration::from_micros(rand::thread_rng().gen_range(0..100));
890        tokio::time::sleep(sleep_time).await;
891
892        // collectors
893        let mut num_requests = 0;
894        let mut num_blocked = 0;
895        let mut time_to_first_block = None;
896        let mut total_time_blocked = Duration::from_micros(0);
897        let mut num_blocklist_adds = 0;
898        // state variables
899        let mut currently_blocked = false;
900        let mut time_blocked_start = Instant::now();
901        let start = Instant::now();
902
903        // we use ticker instead of sleep to be as close to the target TPS as possible,
904        let sleep_time = Duration::from_micros(1_000_000 / per_client_tps as u64);
905        let mut interval_ticker = time::interval(sleep_time);
906
907        while start.elapsed() < duration {
908            let client = Some(IpAddr::V4(Ipv4Addr::new(127, 0, 0, task_num)));
909            let allowed = controller.check(&client, &None).await;
910            if allowed {
911                if currently_blocked {
912                    total_time_blocked += time_blocked_start.elapsed();
913                    currently_blocked = false;
914                }
915                controller.tally(TrafficTally::new(
916                    client,
917                    // TODO add proxy IP for testing
918                    None,
919                    // TODO add weight adjustments
920                    None,
921                    Weight::one(),
922                ));
923            } else {
924                if !currently_blocked {
925                    time_blocked_start = Instant::now();
926                    currently_blocked = true;
927                    num_blocklist_adds += 1;
928                    if time_to_first_block.is_none() {
929                        time_to_first_block = Some(start.elapsed());
930                    }
931                }
932                num_blocked += 1;
933            }
934            num_requests += 1;
935
936            interval_ticker.tick().await;
937        }
938        TrafficSimMetrics {
939            num_requests,
940            num_blocked,
941            time_to_first_block,
942            abs_time_to_first_block: time_to_first_block,
943            total_time_blocked,
944            num_blocklist_adds,
945        }
946    }
947
948    fn report_metrics(
949        metrics: TrafficSimMetrics,
950        duration: Duration,
951        per_client_tps: usize,
952        num_clients: u8,
953    ) {
954        println!("TrafficSim metrics:");
955        println!("-------------------");
956        // The below two should be near equal
957        println!(
958            "Num expected requests: {}",
959            per_client_tps * (num_clients as usize) * duration.as_secs() as usize
960        );
961        println!("Num actual requests: {}", metrics.num_requests);
962        // This reflects the number of requests that were blocked, but note that once a
963        // client is added to the blocklist, all subsequent requests from that
964        // client are blocked until ttl is expired.
965        println!("Num blocked requests: {}", metrics.num_blocked);
966        // This metric on the other hand reflects the number of times a client was added
967        // to the blocklist and thus can be compared an the expectation based on
968        // the policy block threshold and ttl
969        println!(
970            "Num times added to blocklist: {}",
971            metrics.num_blocklist_adds
972        );
973        // This averages the duration for the first request to be blocked across all
974        // clients, which is useful for understanding if the policy is rate
975        // limiting based on expectation
976        let avg_first_block_time = metrics
977            .time_to_first_block
978            .map(|ttf| ttf / num_clients as u32);
979        println!("Average time to first block: {avg_first_block_time:?}");
980        // This is the time it took for the first request to be blocked across all
981        // clients, and is instead more useful for understanding false positives
982        // in terms of rate and magnitude.
983        println!(
984            "Absolute time to first block (across all clients): {:?}",
985            metrics.abs_time_to_first_block
986        );
987        // Useful for ensuring that TTL is respected
988        let avg_time_blocked = (metrics.total_time_blocked.as_millis() as u64)
989            .checked_div(metrics.num_blocklist_adds)
990            .unwrap_or_default();
991
992        println!(
993            "Average time blocked (ttl): {:?}",
994            Duration::from_millis(avg_time_blocked)
995        );
996    }
997}

999pub fn parse_ip(ip: &str) -> Option<IpAddr> {
1000    ip.parse::<IpAddr>().ok().or_else(|| {
1001        ip.parse::<SocketAddr>()
1002            .ok()
1003            .map(|socket_addr| socket_addr.ip())
1004            .or_else(|| {
1005                error!("Failed to parse value of {:?} to ip address or socket.", ip,);
1006                None
1007            })
1008    })
